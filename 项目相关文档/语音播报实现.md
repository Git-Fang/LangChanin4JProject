## 二、实现方案概述

### 2.1 技术选型

- **Web Speech API**: 浏览器原生API，无需额外依赖
  - `SpeechSynthesisUtterance`: 语音合成接口
  - `speechSynthesis`: 全局语音合成管理器
- **优势**: 轻量级、跨浏览器支持、无需后端支持

### 2.2 功能设计

- **按钮位置**: 与复制/重试按钮并排放置
- **图标建议**: 🔊 (speaker) 或 🔠 (text-to-speech)
- **功能**: 将消息区域内的文字内容转换为语音播放
- **音量**: 使用系统默认音量设置

## 三、详细实现步骤

### 3.1 统一工具模块设计

**创建文件**: `src/main/resources/static/js/text-to-speech.js`

```javascript
/**
 * 文本转语音工具类
 * 封装Web Speech API，提供统一的语音播报功能
 */
const TextToSpeech = {
    synth: null,
    currentUtterance: null,

    /**
     * 初始化语音合成器
     */
    init() {
        if (!this.synth) {
            this.synth = window.speechSynthesis;
        }
        return this.synth !== null;
    },

    /**
     * 检查浏览器是否支持语音合成
     * @returns {boolean}
     */
    isSupported() {
        return 'speechSynthesis' in window;
    },

    /**
     * 获取可用的语音列表
     * @returns {Array<SpeechSynthesisVoice>}
     */
    getVoices() {
        if (!this.init()) return [];
        return this.synth.getVoices();
    },

    /**
     * 播报文本内容
     * @param {string} text - 要播报的文本内容
     * @param {Object} options - 配置选项
     * @param {string} options.lang - 语言设置，默认 'zh-CN'
     * @param {number} options.rate - 语速，0.1-10，默认 1
     * @param {number} options.pitch - 音调，0-2，默认 1
     * @param {number} options.volume - 音量，0-1，默认 1
     */
    speak(text, options = {}) {
        if (!this.isSupported()) {
            console.warn('浏览器不支持语音合成功能');
            return false;
        }

        if (!text || text.trim() === '') {
            console.warn('没有可播报的内容');
            return false;
        }

        // 停止当前正在播放的语音
        this.stop();

        // 清理文本内容（移除Markdown格式等）
        const cleanText = this.cleanText(text);

        const defaultOptions = {
            lang: 'zh-CN',
            rate: 1,
            pitch: 1,
            volume: 1
        };

        const config = { ...defaultOptions, ...options };

        const utterance = new SpeechSynthesisUtterance(cleanText);
        utterance.lang = config.lang;
        utterance.rate = config.rate;
        utterance.pitch = config.pitch;
        utterance.volume = config.volume;

        // 事件处理
        utterance.onstart = () => {
            console.log('语音播报开始');
        };

        utterance.onend = () => {
            console.log('语音播报完成');
        };

        utterance.onerror = (event) => {
            console.error('语音播报错误:', event.error);
        };

        this.currentUtterance = utterance;
        this.synth.speak(utterance);

        return true;
    },

    /**
     * 停止当前语音播报
     */
    stop() {
        if (this.synth) {
            this.synth.cancel();
            this.currentUtterance = null;
        }
    },

    /**
     * 暂停语音播报
     */
    pause() {
        if (this.synth) {
            this.synth.pause();
        }
    },

    /**
     * 恢复语音播报
     */
    resume() {
        if (this.synth) {
            this.synth.resume();
        }
    },

    /**
     * 检查是否正在播报
     * @returns {boolean}
     */
    isSpeaking() {
        return this.synth ? this.synth.speaking : false;
    },

    /**
     * 清理文本内容
     * 移除Markdown格式、多余空白等
     * @param {string} text
     * @returns {string}
     */
    cleanText(text) {
        if (!text) return '';

        // 移除Markdown代码块标记
        let cleaned = text.replace(/```[\s\S]*?```/g, (match) => {
            return match.replace(/```\w*\n?/g, '').replace(/```/g, '');
        });

        // 移除行内代码标记
        cleaned = cleaned.replace(/`([^`]+)`/g, '$1');

        // 移除粗体标记
        cleaned = cleaned.replace(/\*\*([^*]+)\*\*/g, '$1');
        cleaned = cleaned.replace(/__([^_]+)__/g, '$1');

        // 移除斜体标记
        cleaned = cleaned.replace(/\*([^*]+)\*/g, '$1');
        cleaned = cleaned.replace(/_([^_]+)_/g, '$1');

        // 移除链接，保留文本
        cleaned = cleaned.replace(/\[([^\]]+)\]\([^)]+\)/g, '$1');

        // 移除图片
        cleaned = cleaned.replace(/!\[([^\]]*)\]\([^)]+\)/g, '$1');

        // 移除列表标记
        cleaned = cleaned.replace(/^[\s]*[-*+][\s]+/gm, '');
        cleaned = cleaned.replace(/^[\s]*\d+[\.。)][\s]+/gm, '');

        // 移除标题标记
        cleaned = cleaned.replace(/^#{1,6}[\s]+/gm, '');

        // 移除引用标记
        cleaned = cleaned.replace(/^[\s]*>[>\s]*/gm, '');

        // 移除水平线
        cleaned = cleaned.replace(/^[-*_]{3,}$/gm, '');

        // 移除HTML标签
        cleaned = cleaned.replace(/<[^>]+>/g, '');

        // 规范化空白字符
        cleaned = cleaned.replace(/\s+/g, ' ').trim();

        return cleaned;
    }
};

// 初始化语音引擎
TextToSpeech.init();
```

```javascript
/**
 * 创建语音播报按钮
 * @param {HTMLElement} actionsContainer - 按钮容器
 * @param {string} content - 要播报的内容
 * @param {HTMLElement} messageElement - 消息元素（用于状态管理）
 */
function createTTSButton(actionsContainer, content, messageElement) {
    const ttsBtn = document.createElement('button');
    ttsBtn.className = 'message-action-btn tts-btn';
    ttsBtn.innerHTML = '<span>🔊</span> 播报';
    ttsBtn.onclick = () => toggleSpeech(content, ttsBtn);
    actionsContainer.appendChild(ttsBtn);
}

/**
 * 切换语音播报状态
 * @param {string} content - 要播报的内容
 * @param {HTMLElement} button - 按钮元素
 */
function toggleSpeech(content, button) {
    if (TextToSpeech.isSpeaking()) {
        TextToSpeech.stop();
        button.classList.remove('speaking');
        button.innerHTML = '<span>🔊</span> 播报';
    } else {
        TextToSpeech.speak(content, {
            lang: 'zh-CN',
            rate: 1,
            pitch: 1,
            volume: 1
        });
        button.classList.add('speaking');
        button.innerHTML = '<span>🔊</span> 停止';
    }

    // 监听语音结束事件，自动恢复按钮状态
    const originalOnEnd = TextToSpeech.currentUtterance?.onend;
    if (TextToSpeech.currentUtterance) {
        TextToSpeech.currentUtterance.onend = function() {
            button.classList.remove('speaking');
            button.innerHTML = '<span>🔊</span> 播报';
            if (originalOnEnd) originalOnEnd.call(this);
        };
    }
}
```

## 四、启动配置说明

### 4.1 deploy-desktop.bat启动

**影响**: 无需修改

**原因**:
- `deploy-desktop.bat` 只负责后端Docker环境搭建
- 前端资源通过Spring Boot静态资源服务提供
- 新增的`js/text-to-speech.js`文件会自动被加载

**影响**: 无需修改

**原因**:
- IDEA启动Spring Boot应用
- 前端资源从`src/main/resources/static/`目录提供
- 新增文件会自动被热加载

## 六、兼容性说明

### 6.1 浏览器支持

- **Chrome**: 完整支持 ✓
- **Firefox**: 完整支持 ✓
- **Edge**: 完整支持 ✓
- **Safari**: 部分支持（可能需要用户交互触发）
- **IE**: 不支持

### 6.2 降级策略

- 当浏览器不支持Web Speech API时，按钮功能自动隐藏
- 控制台输出警告信息，不影响其他功能

### 6.3 语言支持

- 默认使用 `zh-CN`（中文）
- 可通过 `TextToSpeech.getVoices()` 获取可用语音列表

## 七、注意事项

1. **用户隐私**: 首次使用可能需要用户交互才能播放音频
2. **自动播放限制**: 浏览器可能阻止自动播放，需要用户先与页面交互
3. **资源清理**: 页面关闭时调用 `TextToSpeech.stop()` 释放资源
4. **Markdown处理**: `cleanText()` 方法已处理常见Markdown格式，但复杂嵌套格式可能需要额外处理
5. **性能考虑**: 大量文本时，建议限制单次播报长度或提供分页播报功能

---
